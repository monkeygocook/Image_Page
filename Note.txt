เบลอเฉพาะบางส่วน
ช้ clip()— เหมือนเอากระดาษเจาะรูมาทาบแล้วพ่นสี
ctx.filter = "blur(20px)";
ctx.drawImage(img, ...);        // 1. เบลอทั้งภาพก่อน
ctx.filter = "none";

ctx.save();                     // 2. จำสถานะไว้
ctx.beginPath();
ctx.ellipse(...);               // 3. วาดวงรี = "รู" ที่จะพ่นผ่าน
ctx.clip();                     // 4. ต่อไปนี้วาดได้เฉพาะในวงรี
ctx.drawImage(img, ...);        // 5. ทับด้วยภาพคม → กลางชัด ขอบเบลอ
ctx.restore();                  // 6. คืนสถานะ ยกเลิก clip
